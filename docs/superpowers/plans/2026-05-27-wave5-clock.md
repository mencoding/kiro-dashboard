# Wave 5 / Frente L — Clock injetável Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Tornar todas as funções de janela temporal do `aggregator` (e correlatos) determinísticas via parâmetro opcional `now: datetime | None`. Quando ausente, usa `datetime.now(timezone.utc)` como hoje. Substitui o hack v0.5.1 (offsets curtos) por testes 100% timezone-safe. Base arquitetural pra Wave 5 (snapshot, history, comparativos).

**Architecture:**

- **DI puro, sem dep:** cada função que precisa do "agora" aceita `now: datetime | None = None` como kwarg. `_resolve_now(now)` helper retorna `now` ou `datetime.now(timezone.utc)`.
- **Não quebra API existente:** todas as chamadas atuais sem o kwarg continuam funcionando. Migração interna gradual.
- **Compat com Frentes seguintes:** Frentes M/N/O dependem dessa injeção — snapshots calculam "ontem" e a Frente N (queries históricas) precisa de "now" estável pra resolver "month corrente" / "year corrente".

**Tech Stack:** Python 3.12 stdlib. Sem deps novas.

**Branch:** `feat/wave5-clock`

def _resolve_now(now: datetime | None) -> datetime:
    """Retorna ``now`` se passado, senão ``datetime.now(timezone.utc)``.

    Centralizar o ponto de injeção facilita a testabilidade timezone-safe
    e a futura introdução de Clock alternativos (snapshot, replay).
    """
    return now if now is not None else datetime.now(timezone.utc)


def turns_in_local_day(
    sessions: list[Session],
    d: date | None = None,
    *,
    now: datetime | None = None,
) -> list[tuple[Session, Turn]]:
    """Pares (sessão, turn) do dia ``d`` (local).

    ``d`` default = "hoje" computado a partir de ``now`` (default
    ``datetime.now(timezone.utc)``). ``now`` permite testes determinísticos.
    """
    if d is None:
        n = _resolve_now(now)
        d = n.astimezone().date()
    start_utc, end_utc = _local_day_bounds(d)

    out: list[tuple[Session, Turn]] = []
    for s in sessions:
        for t in s.turns_in(start_utc, end_utc):
            out.append((s, t))
    return out
```

def turns_in_last_days(
    sessions: list[Session],
    days: int,
    *,
    now: datetime | None = None,
) -> list[tuple[Session, Turn]]:
    if days <= 0:
        return []
    n = _resolve_now(now)
    cutoff = n - timedelta(days=days)
    out = []
    for s in sessions:
        for t in s.turns:
            if t.end_timestamp is None:
                continue
            if cutoff <= t.end_timestamp <= n:
                out.append((s, t))
    return out


def turns_in_cycle(
    sessions: list[Session],
    cycle_start: date,
    *,
    now: datetime | None = None,
) -> list[tuple[Session, Turn]]:
    n = _resolve_now(now)
    # Início do ciclo no fuso local
    tz_local = datetime.now().astimezone().tzinfo
    start_local = datetime.combine(cycle_start, time.min, tzinfo=tz_local)
    start_utc = start_local.astimezone(timezone.utc)

    out = []
    for s in sessions:
        for t in s.turns:
            if t.end_timestamp is None:
                continue
            if start_utc <= t.end_timestamp <= n:
                out.append((s, t))
    return out


def resolve_window(
    sessions: list[Session],
    window: str,
    *,
    cycle_start: date,
    now: datetime | None = None,
) -> list[tuple[Session, Turn]]:
    w = window.strip().lower()
    if w == "today":
        return turns_in_local_day(sessions, now=now)
    if w == "week":
        return turns_in_last_days(sessions, days=7, now=now)
    if w == "month":
        return turns_in_last_days(sessions, days=30, now=now)
    if w == "cycle":
        return turns_in_cycle(sessions, cycle_start, now=now)
    if w == "all":
        return [(s, t) for s in sessions for t in s.turns]
    try:
        n_days = int(w)
    except ValueError as exc:
        raise ValueError(f"window inválido: '{window}'.") from exc
    if n_days < 0:
        raise ValueError(f"window negativo: {n_days}")
    return turns_in_last_days(sessions, days=n_days, now=now)
```
